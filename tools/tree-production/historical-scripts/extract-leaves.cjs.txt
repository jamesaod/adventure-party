const fs=require('fs'),p=require('path'),root=__dirname;let v=[],uv=[],n=[],faces=[],mat='';for(const l of fs.readFileSync(p.join(root,'WhiteOak_LeafFall.obj'),'utf8').split('\n')){const a=l.trim().split(/\s+/),t=a.shift();if(t==='v')v.push(a.map(Number));if(t==='vt')uv.push(a.map(Number));if(t==='vn')n.push(a.map(Number));if(t==='usemtl')mat=a[0];if(t==='f'&&!mat.includes('Bark'))faces.push({mat,verts:a.map(s=>s.split('/').map(Number))});}const parents=faces.map((_,i)=>i),find=i=>parents[i]===i?i:parents[i]=find(parents[i]),seen=new Map;faces.forEach((f,i)=>f.verts.forEach(x=>{const key=f.mat+':'+v[x[0]-1].join(',');if(seen.has(key))parents[find(i)]=find(seen.get(key));else seen.set(key,i)}));const groups=new Map;faces.forEach((f,i)=>{const k=find(i);if(!groups.has(k))groups.set(k,[]);groups.get(k).push(f)});const comps=[...groups.values()].map(ff=>{const ids=[...new Set(ff.flatMap(f=>f.verts.map(x=>x[0]-1)))],pts=ids.map(i=>v[i]);return {faces:ff,mat:ff[0].mat,min:[0,1,2].map(k=>Math.min(...pts.map(x=>x[k]))),max:[0,1,2].map(k=>Math.max(...pts.map(x=>x[k])))};});console.log(comps.filter(c=>c.max[2]<2).map(c=>({mat:c.mat,f:c.faces.length,z:[c.min[2],c.max[2]]})).slice(0,30));fs.writeFileSync(p.join(root,'leaf-extract-data.json'),JSON.stringify({v,uv,n,comps:comps.filter(c=>c.max[2]<2&&c.max[2]>0)}));
