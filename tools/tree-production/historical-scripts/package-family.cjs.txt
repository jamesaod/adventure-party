const fs=require('fs'),p=require('path');const root=__dirname,dest=p.join(root,'comparison/assets/family');fs.mkdirSync(dest,{recursive:true});let report=[];
for(const name of ['RaisedCrown','Fall','Winter','Fallen','Stump']){const base='WhiteOak_'+name;const mtl=fs.readFileSync(p.join(root,base+'.mtl'),'utf8');for(const f of new Set([base+'.mtl',...Array.from(mtl.matchAll(/^map_\w+ (.+)$/gm),m=>m[1].trim())]))fs.copyFileSync(p.join(root,f),p.join(dest,f));let lods=[];for(let i=0;i<6;i++){const file=base+(i?'_LOD'+i:'')+'.obj';fs.copyFileSync(p.join(root,file),p.join(dest,file));const txt=fs.readFileSync(p.join(root,file),'utf8');let v=[],count=0,by={},mat='';for(const l of txt.split('\n')){const a=l.trim().split(/\s+/);if(a[0]==='v')v.push(a.slice(1).map(Number));if(a[0]==='usemtl')mat=a[1];if(a[0]==='f'){count+=a.length-3;by[mat]=(by[mat]||0)+a.length-3;}}const min=[0,1,2].map(k=>Math.min(...v.map(x=>x[k]))),max=[0,1,2].map(k=>Math.max(...v.map(x=>x[k])));lods.push({file,triangles:count,min,max,woodyOutside2mMinZ:Math.min(...v.filter(x=>Math.hypot(x[0],x[1])>2&&x[2]>1.5).map(x=>x[2]))-.2,materials:by});}report.push({name,base,lods});}fs.writeFileSync(p.join(dest,'manifest.json'),JSON.stringify(report,null,2));console.log(report.map(r=>({name:r.name,triangles:r.lods.map(x=>x.triangles),height:r.lods[0].max[2]-.2,clearance:r.lods[0].woodyOutside2mMinZ})));
