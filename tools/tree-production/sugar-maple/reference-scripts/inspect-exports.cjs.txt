const fs=require('fs'),path=require('path');
const dir=path.resolve('outputs/sugar-maple');
for(const file of fs.readdirSync(dir).filter(x=>x.endsWith('.obj'))){let lo=[Infinity,Infinity,Infinity],hi=[-Infinity,-Infinity,-Infinity],faces=0,vs=0,mats={},mat='';for(const l of fs.readFileSync(path.join(dir,file),'utf8').split(/\r?\n/)){if(l.startsWith('v ')){const v=l.split(/\s+/).slice(1).map(Number);v.forEach((x,i)=>{lo[i]=Math.min(lo[i],x);hi[i]=Math.max(hi[i],x)});vs++}else if(l.startsWith('usemtl '))mat=l.slice(7);else if(l.startsWith('f ')){const n=l.trim().split(/\s+/).length-3;faces+=n;mats[mat]=(mats[mat]||0)+n}}console.log(JSON.stringify({file,vertices:vs,triangles:faces,materials:mats,min:lo,max:hi,size:hi.map((x,i)=>x-lo[i])}));}
